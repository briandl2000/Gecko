find_program(GECKO_GLSLC glslc HINTS $ENV{VULKAN_SDK}/bin)
if(NOT GECKO_GLSLC)
  message(STATUS "graphics_example: glslc not found — example disabled. "
                 "Install the Vulkan SDK or set VULKAN_SDK to enable.")
  return()
endif()

add_executable(graphics_example
  src/main.cpp
)

target_link_libraries(graphics_example
  PRIVATE
    Gecko::Core
    Gecko::Runtime
    Gecko::Platform
    Gecko::Graphics
)

set(_shader_src_dir "${CMAKE_CURRENT_SOURCE_DIR}/shaders")
set(_shader_out_dir "${CMAKE_CURRENT_BINARY_DIR}/shaders")
file(MAKE_DIRECTORY "${_shader_out_dir}")

# Each entry packs the four fields into a single token separated by `|`
# (CMake lists use `;` natively, so an inner `|` keeps the tuple intact).
#   <input.hlsl>|<stage>|<output.spv>|<extra_flags>
set(_shader_list
  "triangle.vert.hlsl|vert|triangle.vert.spv|"
  "triangle.frag.hlsl|frag|triangle.frag.spv|"
  "fullscreen.vert.hlsl|vert|fullscreen.vert.spv|"
  "fullscreen.frag.hlsl|frag|fullscreen.frag.spv|"
  "plasma.comp.hlsl|comp|plasma.comp.spv|"
)

set(_spv_outputs "")
foreach(_entry IN LISTS _shader_list)
  string(REPLACE "|" ";" _fields "${_entry}")
  list(GET _fields 0 _src_name)
  list(GET _fields 1 _stage)
  list(GET _fields 2 _out_name)
  list(GET _fields 3 _extra)
  set(_in  "${_shader_src_dir}/${_src_name}")
  set(_out "${_shader_out_dir}/${_out_name}")
  set(_extra_args "")
  if(_extra)
    set(_extra_args "${_extra}")
  endif()
  add_custom_command(
    OUTPUT  "${_out}"
    COMMAND "${GECKO_GLSLC}"
            -x hlsl
            -fshader-stage=${_stage}
            -fentry-point=main
            ${_extra_args}
            "${_in}"
            -o "${_out}"
    DEPENDS "${_in}"
    COMMENT "glslc ${_src_name} -> ${_out_name}"
    VERBATIM
  )
  list(APPEND _spv_outputs "${_out}")
endforeach()

# Generate shaders.h that #embeds each .spv.
set(_shaders_header "${_shader_out_dir}/shaders.h")
set(_gen_script "${CMAKE_CURRENT_BINARY_DIR}/gen_shaders_header.cmake")
file(WRITE "${_gen_script}" [[
# Auto-generated by examples/graphics_example/CMakeLists.txt
file(WRITE "${OUT}"
"// Auto-generated — do not edit.\n"
"// Shader SPIR-V blobs embedded via C++26 #embed.\n"
"#pragma once\n"
"\n"
"namespace gecko::graphics_example::shaders {\n"
"\n"
"#if defined(__clang__)\n"
"#  pragma clang diagnostic push\n"
"#  pragma clang diagnostic ignored \"-Wc23-extensions\"\n"
"#endif\n"
"\n"
"alignas(4) inline constexpr unsigned char TriangleVert[] = {\n"
"#embed \"triangle.vert.spv\"\n"
"};\n"
"alignas(4) inline constexpr unsigned char TriangleFrag[] = {\n"
"#embed \"triangle.frag.spv\"\n"
"};\n"
"alignas(4) inline constexpr unsigned char FullscreenVert[] = {\n"
"#embed \"fullscreen.vert.spv\"\n"
"};\n"
"alignas(4) inline constexpr unsigned char FullscreenFrag[] = {\n"
"#embed \"fullscreen.frag.spv\"\n"
"};\n"
"alignas(4) inline constexpr unsigned char PlasmaComp[] = {\n"
"#embed \"plasma.comp.spv\"\n"
"};\n"
"\n"
"#if defined(__clang__)\n"
"#  pragma clang diagnostic pop\n"
"#endif\n"
"\n"
"}  // namespace gecko::graphics_example::shaders\n"
)
]])

add_custom_command(
  OUTPUT  "${_shaders_header}"
  COMMAND ${CMAKE_COMMAND} -DOUT=${_shaders_header} -P "${_gen_script}"
  DEPENDS ${_spv_outputs} "${_gen_script}"
  COMMENT "Generating shaders.h"
  VERBATIM
)

add_custom_target(graphics_example_shaders
  DEPENDS "${_shaders_header}"
)
add_dependencies(graphics_example graphics_example_shaders)

target_include_directories(graphics_example PRIVATE "${_shader_out_dir}")
# Embed search path so `#embed "triangle.vert.spv"` inside the generated
# header resolves to the compiled blobs.
target_compile_options(graphics_example PRIVATE --embed-dir=${_shader_out_dir})
